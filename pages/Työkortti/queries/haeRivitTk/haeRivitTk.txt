SELECT r.id, COALESCE(r.materiaali_id::text, 'oma') AS materiaali, r.kuvaus, COALESCE(r.leveys_mm, m.leveys_mm) AS leveys_mm, COALESCE(r.yksikko, m.kayttoyksikko) AS yksikko, r.maara_arvio, r.maara_toteuma, r.lukittu, r.jarjestys FROM public.rivit r LEFT JOIN public.materiaalit m ON m.id = r.materiaali_id WHERE r.toimeksianto_id = {{appsmith.URL.queryParams.id}}::uuid AND r.rivityyppi IN ('materiaali','oma_materiaali') ORDER BY r.jarjestys, r.created_at;
