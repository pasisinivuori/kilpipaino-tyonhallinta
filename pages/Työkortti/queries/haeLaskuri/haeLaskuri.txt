SELECT jsonb_build_object(
 'haettu', extract(epoch from clock_timestamp()),
 'tid', p.tid::text,
 'asetukset', (SELECT jsonb_build_object('tv',a.tuntiveloitus,'tk',a.tunnin_kustannus,'kate',a.tyon_oletuskate_pros,'hukka',a.oletus_hukka_pros,'pikku',a.pikkutyon_raja,'reuna',a.rullan_reunavara_mm,'vali',a.kappalevali_mm) FROM public.asetukset a WHERE a.id = 1),
 'materiaalit', (SELECT coalesce(jsonb_agg(jsonb_build_object('id',m.id,'n',m.nimi,'t',mt.nimi,'ky',m.kayttoyksikko,'oy',m.ostoyksikko,'ek',m.ostoeran_koko,'hh',m.hankintahinta,'kh',m.kayttoyksikon_hinta,'lev',m.leveys_mm,'pit',m.pituus_mm,'hukka',m.hukka_pros,'kate',m.kate_pros,'akt',m.aktiivinen,'k',(SELECT coalesce(jsonb_agg(mk.kone_id),'[]'::jsonb) FROM public.materiaali_koneet mk WHERE mk.materiaali_id = m.id)) ORDER BY mt.jarjestys NULLS LAST, mt.nimi NULLS LAST, m.nimi, m.leveys_mm NULLS FIRST),'[]'::jsonb) FROM public.materiaalit m LEFT JOIN public.materiaalityypit mt ON mt.id = m.tyyppi_id),
 'koneet', (SELECT coalesce(jsonb_agg(jsonb_build_object('id',k.id,'n',k.kone,'kust',k.kustannus_per_tunti,'km',k.kustannus_per_m,'km2',k.kustannus_per_m2,'aste',k.kayttoaste_pros,'kate',k.kate_pros,'akt',k.aktiivinen) ORDER BY k.kone),'[]'::jsonb) FROM public.koneet k),
 'tuotteet', (SELECT coalesce(jsonb_agg(jsonb_build_object('id',t.id,'n',t.nimi,'y',t.yksikko,'mh',t.myyntihinta,'ok',t.omakustannus,'akt',t.aktiivinen) ORDER BY t.nimi),'[]'::jsonb) FROM public.tuotteet t),
 'tyo', (SELECT jsonb_build_object('tarjous',x.hinta_yhteensa,'perus',x.laskutusperuste,'tp',x.tp_numero) FROM public.toimeksiannot x WHERE x.id = p.tid),
 'rivit', (SELECT coalesce(jsonb_agg(jsonb_build_object('id',r.id,'tyyppi',r.rivityyppi,'materiaali_id',r.materiaali_id,'kone_id',r.kone_id,'tuote_id',r.tuote_id,'kuvaus',r.kuvaus,'yksikko',r.yksikko,'maara',r.maara_arvio,'toteuma',r.maara_toteuma,'osto',r.maara_ostoyksikkoina,'yksikkohinta',r.yksikkohinta,'omakustannus',r.omakustannus,'hukka',r.hukka_pros,'kate',r.kate_pros,'syottotapa',r.syottotapa,'w',r.mitta_leveys_mm,'h',r.mitta_korkeus_mm,'kpl',r.kpl,'vali',r.vali_mm,'kaanna',r.saa_kaantaa,'nayta',r.nayta_tarjouksessa,'linkki',r.linkki_rivi_id,'seuraa',r.maara_seuraa,'jarjestys',r.jarjestys,'lukittu',r.lukittu,'h_yks',h.yksikko,'h_yh',h.yksikkohinta,'h_hinta',h.rivihinta,'h_hintat',h.rivihinta_toteuma,'h_kust',h.omakustannus,'h_kustt',h.omakustannus_toteuma,'h_kate',h.kate_pros,'h_hukka',h.hukka_pros) ORDER BY r.jarjestys, r.created_at),'[]'::jsonb) FROM public.rivit r JOIN public.rivit_hinnoiteltu h ON h.id = r.id WHERE r.toimeksianto_id = p.tid)
) AS d
FROM (SELECT NULLIF({{appsmith.URL.queryParams.id || ''}}, '')::uuid AS tid) p;
