SELECT lahde, pvm::text AS pvm, to_char(alkaa,'HH24:MI') AS alkaa, to_char(paattyy,'HH24:MI') AS paattyy, otsikko, lisatieto, vari, toimeksianto_id::text AS toimeksianto_id, merkinta_id::text AS merkinta_id, tp_numero FROM public.kalenteri_tapahtumat WHERE pvm >= (date_trunc('month', {{ appsmith.store.kuuAnkkuri || moment().format("YYYY-MM-DD") }}::date) - interval '7 days') AND pvm < (date_trunc('month', {{ appsmith.store.kuuAnkkuri || moment().format("YYYY-MM-DD") }}::date) + interval '1 month' + interval '7 days') ORDER BY pvm, alkaa NULLS FIRST, lahde, otsikko;
