SELECT tp_numero, COALESCE(NULLIF(btrim(asiakas_yritys),''), NULLIF(btrim(asiakas_nimi),''), asiakas_email, '') AS asiakas, to_char(valmistumispaiva, 'DD.MM.YYYY') AS valmistumispaiva, (current_date - valmistumispaiva)::int AS myohassa_pv, COALESCE(NULLIF(btrim(omistaja),''),'') AS vastuuhenkilo FROM public.toimeksiannot WHERE arkistoitu=false AND tyyppi='tilaus' AND valmistumispaiva IS NOT NULL AND valmistumispaiva < current_date AND status NOT IN ('valmis','toimitettu','laskutettu') ORDER BY valmistumispaiva ASC;
