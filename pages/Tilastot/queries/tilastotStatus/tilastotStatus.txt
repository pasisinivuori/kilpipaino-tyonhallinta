SELECT (CASE t.status WHEN 'uusi' THEN 'Uusi' WHEN 'tarjous_lahetetty' THEN 'Tarjous lähetetty' WHEN 'tilaus_vastaanotettu' THEN 'Tilaus vastaanotettu' WHEN 'suunnittelussa' THEN 'Suunnittelussa' WHEN 'tuotannossa' THEN 'Tuotannossa' WHEN 'odottaa_asennusta' THEN 'Odottaa asennusta' ELSE t.status::text END) || ' (' || COUNT(*)::text || ' kpl)' AS x, COUNT(*)::int AS y FROM public.toimeksiannot t WHERE t.arkistoitu = false AND t.status NOT IN ('valmis','toimitettu','laskutettu','hylatty') GROUP BY t.status ORDER BY array_position(ARRAY['uusi','tarjous_lahetetty','tilaus_vastaanotettu','suunnittelussa','tuotannossa','odottaa_asennusta']::text[], t.status::text);
