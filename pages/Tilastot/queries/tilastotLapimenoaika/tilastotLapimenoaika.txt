SELECT round(avg(extract(epoch from (l.t1 - a.t0))/86400)::numeric, 1) AS keskiarvo_pv, count(*)::int AS n FROM (SELECT toimeksianto_id, min(created_at) AS t0 FROM public.tapahtumat WHERE uusi_status='tilaus_vastaanotettu' GROUP BY toimeksianto_id) a JOIN (SELECT toimeksianto_id, min(created_at) AS t1 FROM public.tapahtumat WHERE uusi_status IN ('valmis','toimitettu') GROUP BY toimeksianto_id) l ON l.toimeksianto_id=a.toimeksianto_id WHERE l.t1 >= a.t0;
