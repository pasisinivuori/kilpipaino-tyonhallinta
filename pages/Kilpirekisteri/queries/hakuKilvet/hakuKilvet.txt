select k.id, t.id as taloyhtio_id, t.nimi as taloyhtio, (select string_agg(trim(concat_ws(' ', o.katu, o.katunumero)) || coalesce(', ' || nullif(trim(concat_ws(' ', o.postinumero, o.kaupunki)), ''), ''), ' / ' order by o.jarjestys) from kilvet.osoite o where o.taloyhtio_id = t.id) as osoite, k.tuotelaji, coalesce(k.tyyppi, k.koko) as kilpi, k.vari, k.kiinnitys, k.materiaali, k.pintakasittely, k.malli_tiedosto, case when k.tarkistettava then 'Tarkistettava' when k.tila = 'taydennetaan_tilattaessa' then 'Taydennetaan tilattaessa' else 'Kaytettavissa' end as tila, k.tarkistuksen_syy, k.muistiinpano, k.kiinnitys_paatelty, k.materiaali_paatelty, k.excel_rivi from kilvet.kilpi k join kilvet.taloyhtio t on t.id = k.taloyhtio_id order by t.nimi nulls last, k.tuotelaji;
