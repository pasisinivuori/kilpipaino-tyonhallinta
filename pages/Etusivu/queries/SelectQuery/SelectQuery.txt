SELECT t.id, t.tp_numero, t.tyyppi, t.status, t.omistaja, t.aihe, COALESCE(t.asiakas_nimi, t.asiakas_email) AS asiakas, t.asiakas_yritys AS yritys, t.asiakas_nimi AS yhteyshenkilo, t.vastattava_mennessa, t.yhteenveto, t.created_at, COALESCE((SELECT v.suunta = 'saapuva' FROM public.viestit v WHERE v.toimeksianto_id = t.id ORDER BY v.sent_at DESC NULLS LAST, v.created_at DESC LIMIT 1) AND COALESCE((SELECT max(COALESCE(v2.sent_at, v2.created_at)) FROM public.viestit v2 WHERE v2.toimeksianto_id = t.id AND v2.suunta = 'saapuva'), t.created_at) > COALESCE(t.kuitattu_ilman_vastausta_at, '-infinity'::timestamptz), false) AS odottaa_meilta, (t.status = 'uusi' AND now() >= public.arki_maaraaika(t.created_at, 48)) AS saapunut_punainen, (t.uusi_viesti_suljettuun_at IS NOT NULL) AS uusi_viesti_suljettuun, t.statusilmoitukset, (j.on_jonossa IS NOT NULL) AS ilmoitus_jonossa 
     , t.valmistumispaiva
     , CASE
         WHEN t.uusi_viesti_suljettuun_at IS NOT NULL THEN 0
         WHEN t.valmistumispaiva IS NOT NULL AND t.valmistumispaiva < public.tanaan() AND t.status NOT IN ('valmis','toimitettu','laskutettu') THEN 1
         WHEN COALESCE((SELECT v3.suunta = 'saapuva' FROM public.viestit v3 WHERE v3.toimeksianto_id = t.id ORDER BY COALESCE(v3.sent_at, v3.created_at) DESC LIMIT 1), false) THEN 2
         WHEN t.valmistumispaiva IS NOT NULL AND t.valmistumispaiva <= public.tanaan() + 2 AND t.status NOT IN ('valmis','toimitettu','laskutettu') THEN 3
         ELSE 4
       END AS kiire_taso
     , CASE
         WHEN t.uusi_viesti_suljettuun_at IS NOT NULL
           THEN EXTRACT(epoch FROM (now() - t.uusi_viesti_suljettuun_at))/3600
         WHEN t.valmistumispaiva IS NOT NULL AND t.valmistumispaiva < public.tanaan() AND t.status NOT IN ('valmis','toimitettu','laskutettu')
           THEN (public.tanaan() - t.valmistumispaiva) * 24.0
         WHEN COALESCE((SELECT v3.suunta = 'saapuva' FROM public.viestit v3 WHERE v3.toimeksianto_id = t.id ORDER BY COALESCE(v3.sent_at, v3.created_at) DESC LIMIT 1), false)
           THEN GREATEST(COALESCE(EXTRACT(epoch FROM (now() - (SELECT max(COALESCE(v4.sent_at, v4.created_at)) FROM public.viestit v4 WHERE v4.toimeksianto_id = t.id AND v4.suunta = 'saapuva')))/3600, 0),
                         COALESCE(EXTRACT(epoch FROM (now() - t.vastattava_mennessa))/3600, 0))
         WHEN t.valmistumispaiva IS NOT NULL AND t.valmistumispaiva <= public.tanaan() + 2 AND t.status NOT IN ('valmis','toimitettu','laskutettu')
           THEN (2 - (t.valmistumispaiva - public.tanaan())) * 24.0
         ELSE 0
       END AS kiire_odotus
     FROM public.toimeksiannot t LEFT JOIN LATERAL (SELECT 1 AS on_jonossa FROM public.statusilmoitus_jono q WHERE q.toimeksianto_id = t.id AND q.tila = 'odottaa' LIMIT 1) j ON true WHERE (CASE WHEN {{Switch1.isSwitchedOn}} = true
            THEN t.arkistoitu = true
            ELSE (t.arkistoitu = false OR t.uusi_viesti_suljettuun_at IS NOT NULL)
       END) ORDER BY (t.uusi_viesti_suljettuun_at IS NOT NULL) DESC, t.vastattava_mennessa ASC NULLS LAST, t.created_at DESC;
