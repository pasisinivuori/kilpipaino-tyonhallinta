SELECT m.id, m.tyyppi_id, t.nimi AS tyyppi, m.nimi, m.leveys_mm, m.pituus_mm, m.ostoyksikko, m.kayttoyksikko, m.ostoeran_koko, m.hankintahinta, ROUND(m.kayttoyksikon_hinta, 4) AS kayttoyksikon_hinta, m.hukka_pros, m.kate_pros, m.aktiivinen FROM public.materiaalit m LEFT JOIN public.materiaalityypit t ON t.id = m.tyyppi_id ORDER BY t.jarjestys NULLS LAST, t.nimi NULLS LAST, m.nimi, m.leveys_mm NULLS FIRST;
