SELECT id, yritys, nimi, email, puhelin, osoite, COALESCE(yritys, '(ei yritysta)') || CASE WHEN COALESCE(btrim(nimi), '') <> '' THEN ' - ' || nimi ELSE '' END AS nayttonimi FROM public.asiakkaat WHERE COALESCE(yritys, '') ILIKE ('%' || COALESCE({{tHaku.filterText}}, '') || '%') OR COALESCE(nimi, '') ILIKE ('%' || COALESCE({{tHaku.filterText}}, '') || '%') ORDER BY yritys NULLS LAST, nimi LIMIT 50;
